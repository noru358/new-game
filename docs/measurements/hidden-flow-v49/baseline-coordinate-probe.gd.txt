extends SceneTree
var reports := []
func _initialize() -> void: call_deferred("run")
func vector(p: Vector2) -> Array: return [p.x, p.y]
func rect(r: Rect2) -> Array: return [r.position.x,r.position.y,r.size.x,r.size.y]
func run() -> void:
	print("PRIVATE_USERDATA ", OS.get_user_data_dir())
	for name in ["temple_circuit_run", "jungle_south_circuit"]:
		var scene = load("res://game/"+name+".tscn").instantiate()
		scene.profile_save_prefix = "user://"+name+"_probe"
		scene.growth_save_prefix = scene.profile_save_prefix+"_growth"
		root.add_child(scene)
		await process_frame
		scene.set_physics_process(false)
		scene.player.set_physics_process(false)
		var section = scene.temple_section
		var layout = section.layout
		var record := {"scene":name,"entry":rect(layout.ENTRY_TRIGGER),"return":vector(layout.RETURN_POINT),"arrival":vector(layout.FIELD_ENTRY),"exit":rect(layout.EXIT_TRIGGER),"altar":vector(layout.ALTAR),"open":{},"paths":{}}
		for key in ["ENTRY_TRIGGER","RETURN_POINT","FIELD_ENTRY","EXIT_TRIGGER","ALTAR"]:
			var item = layout.get(key)
			var point: Vector2 = item.get_center() if item is Rect2 else item
			record.open[key] = {"clear18":scene.navigation.is_open(point,18),"clear30":scene.navigation.is_open(point,30),"height":scene.terrain.height_at(point)}
		record.paths["main_to_entry"] = scene.navigation.find_path(layout.RETURN_POINT,layout.ENTRY_TRIGGER.get_center()).size()
		record.paths["inside_to_exit"] = scene.navigation.find_path(layout.FIELD_ENTRY,layout.EXIT_TRIGGER.get_center()).size()
		record.paths["inside_to_altar"] = scene.navigation.find_path(layout.FIELD_ENTRY,layout.ALTAR).size()
		scene.player.facing = Vector2.LEFT
		section.enter_garden()
		section.tick(0)
		record["inside"]={"point":vector(scene.player.position),"facing":vector(scene.player.facing),"camera_size":scene.camera.size,"overview_size":scene.overview_camera_size,"bounds":rect(scene.display_bounds()),"hud":section.section_hud.text,"labels":[]}
		for label in section.find_children("*","Label3D",true,false):
			record.inside.labels.append({"text":label.text,"visible":label.visible,"world":[label.position.x,label.position.y,label.position.z]})
		section.tick(1.0)
		record["idle_stays_inside"] = section.in_garden
		section.leave_garden()
		record["return_state"] = {"point":vector(scene.player.position),"bounds":rect(scene.display_bounds()),"facing":vector(scene.player.facing),"camera_size":scene.camera.size,"hud":section.section_hud.text,"labels":[]}
		for label in section.find_children("*","Label3D",true,false):
			record.return_state.labels.append({"text":label.text,"visible":label.visible})
		reports.append(record)
		scene.queue_free()
		await process_frame
	var file := FileAccess.open("res://probe-report.json",FileAccess.WRITE)
	file.store_string(JSON.stringify(reports,"\t"))
	print("PROBE ",JSON.stringify(reports))
	quit()
